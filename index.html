const headers = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type", }; export async function OPTIONS() { return new Response(null, { status: 204, headers }); } export async function POST(req) { const token = process.env.TELEGRAM_BOT_TOKEN; const chatId = process.env.CHAT_ID; if (!token || !chatId) return Response.json({ ok: false }, { status: 500, headers }); const { name, phone, items, total, pay } = await req.json(); if (!name || !phone || !items) return Response.json({ ok: false }, { status: 400, headers }); const text = "🛒 Yangi buyurtma\n\n" + "👤 " + String(name).slice(0, 100) + "\n" + "📞 " + String(phone).slice(0, 30) + "\n" + "📦 " + String(items).slice(0, 1000) + "\n" + "💰 " + Number(total) + " so'm\n" + "💳 " + String(pay).slice(0, 20); const r = await fetch("https://api.telegram.org/bot" + token + "/sendMessage", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ chat_id: chatId, text }), }); return Response.json({ ok: r.ok }, { status: r.ok ? 200 : 502, headers }); }
